def limpiar(texto):
    return texto.strip().capitalize()

def es_valido(nombre):
    if len(nombre) >= 3:
        return True
    return False

nombres = [" bart ", "ED", " walter", "rick "]
validos = []

for nombre in nombres:
    nombre_limpio = limpiar(nombre)

    if es_valido(nombre_limpio):
        validos.append(nombre_limpio)

print(validos)

    ¿Qué hace la función es_valido?
La funcion es_valido hace una comparacion donde el nombre tenga mas de 3 letras si es verdad devuelve True de lo contrario devuelve False
